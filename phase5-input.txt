function isValidCalendarDate(value: string): boolean {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(value)) {
    return false;
  }

  const parsed = new Date(`${value}T00:00:00.000Z`);

  return (
    !Number.isNaN(parsed.getTime()) &&
    parsed.toISOString().slice(0, 10) === value
  );
}

  .string()
  .refine(
    isValidCalendarDate,
    "Enter a valid purchase date in YYYY-MM-DD format"
  )
  .nullable()
  .optional();

  .object({
    sph: z.number().finite().min(-30).max(30).nullable().optional(),
    cyl: z.number().finite().min(-15).max(15).nullable().optional(),
    axis: z.number().int().min(0).max(180).nullable().optional(),
  })
  .strict();

    re_dist: eyeSchema.optional(),
    re_near: eyeSchema.optional(),

    le_near: eyeSchema.optional(),

type DatabaseClient =

type CatalogReference = {
  frameCatalogId?: string | null;
  lensCatalogId?: string | null;
};

type ExistingCatalogSelection = {
  frameCatalogId: string | null;
  lensCatalogId: string | null;
  specsType: string | null;
  lensType: string | null;
};

// This is a business validation failure, not a
// database or server failure.
export class CatalogSelectionError extends Error {
  statusCode = 400;

  constructor(message: string) {
    super(message);
    this.name = "CatalogSelectionError";
  }
}

function checkSnapshotLength(
  name: string,
  label: string
): string {
  if (name.length > 100) {
    throw new CatalogSelectionError(
      `${label} name is too long for a purchase record.`
    );
  }

  return name;
}

// Resolve both catalog references within the same
// transaction used to save the purchase.
export async function resolveCatalogSelection(
  tx: DatabaseClient,
  shopId: string,
  record: RecordInput,
  existing?: ExistingCatalogSelection
) {
  let frameCatalogId =
    record.frameCatalogId !== undefined
      ? record.frameCatalogId
      : existing?.frameCatalogId ?? null;

  let lensCatalogId =
    record.lensCatalogId !== undefined
      ? record.lensCatalogId
      : existing?.lensCatalogId ?? null;

  let specsType =
    record.specsType !== undefined
      ? record.specsType
      : existing?.specsType ?? null;

  let lensType =
    record.lensType !== undefined
      ? record.lensType
      : existing?.lensType ?? null;

  // Newly chosen IDs must refer to active products
  // in the same shop. Previously linked inactive items
  // may remain attached to old purchases.
  if (record.frameCatalogId) {
    const frame = await tx.frameCatalog.findFirst({
      where: {
        id: record.frameCatalogId,
        shopId,
        isActive: true,
      },
      select: {
        id: true,
        brand: true,
        model: true,
      },
    });

    if (!frame) {
      throw new CatalogSelectionError(
        "Selected frame is unavailable or inactive."
      );
    }

    frameCatalogId = frame.id;
    specsType = checkSnapshotLength(
      `${frame.brand} - ${frame.model}`,
      "Frame"
    );
  }

  if (record.lensCatalogId) {
    const lens = await tx.lensCatalog.findFirst({
      where: {
        id: record.lensCatalogId,
        shopId,
        isActive: true,
      },
      select: {
        id: true,
        company: true,
        lensName: true,
        visionType: true,
        coating: true,
      },
    });

    if (!lens) {
      throw new CatalogSelectionError(
        "Selected lens is unavailable or inactive."
      );
    }

    lensCatalogId = lens.id;

    const coating = lens.coating
      ? ` - ${lens.coating}`
      : "";

    lensType = checkSnapshotLength(
      `${lens.company} - ${lens.lensName} - ${lens.visionType}${coating}`,
      "Lens"
    );
  }

  // An explicitly cleared catalog reference may use
  // manual text supplied by the frontend instead.
  return {
    frameCatalogId,
    lensCatalogId,
    specsType,
    lensType,
  };
}

// Use the shop computer's local calendar date.
function todayLocalDate() {
  const now = new Date();

  const year = now.getFullYear();
  const month = String(now.getMonth() + 1).padStart(2, "0");
  const day = String(now.getDate()).padStart(2, "0");

  return `${year}-${month}-${day}`;
}

// Validate real calendar dates, including leap years.
function validDate(value: string) {
  if (!value) return true;

  if (!/^\d{4}-\d{2}-\d{2}$/.test(value)) {
    return false;
  }

  const date = new Date(`${value}T00:00:00.000Z`);

  return (
    !Number.isNaN(date.getTime()) &&
    date.toISOString().slice(0, 10) === value
  );
}

function validMoney(value: string) {
  return (
    /^\d+(\.\d{1,2})?$/.test(value) &&
    Number(value) <= 9999999999.99
  );
}

function money(value: number) {
  return new Intl.NumberFormat("en-IN", {
    style: "currency",
    currency: "INR",
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  }).format(value);
}

}

    le_dist_cyl: z.string(),
    le_dist_axis: z.string(),

    le_near_sph: z.string(),
    le_near_cyl: z.string(),
    le_near_axis: z.string(),
  })
  .superRefine((values, context) => {
    // Only validate the fields used by the pricing mode.
    const priceFields =
      values.pricingMode === "SPLIT"
        ? (["frameRate", "lensRate"] as const)
        : (["mrp"] as const);

    for (const field of priceFields) {
      if (!validMoney(values[field])) {
        context.addIssue({
          code: "custom",
          path: [field],
          message: "Enter a valid price (maximum 2 decimals).",
        });
      }
    }

    // Validate each optical measurement independently.
    for (const eye of eyes) {
      for (const measure of [
        "sph",
        "cyl",
        "axis",
      ] as const) {
        const key: EyeFieldName =
          `${eye.prefix}_${measure}`;

        const value = values[key].trim();

        if (!value) continue;

        const number = Number(value);

        const limit = measure === "sph" ? 30 : 15;

        const valid =
          measure === "axis"
            ? /^\d+$/.test(value) && number <= 180
            : /^[+-]?\d+(\.\d+)?$/.test(value) &&
              Number.isFinite(number) &&
              Math.abs(number) <= limit;

        if (!valid) {
          context.addIssue({
            code: "custom",
            path: [key],
            message:
              measure === "axis"
                ? "AXIS must be between 0 and 180."
                : `${measure.toUpperCase()} must be between -${limit} and +${limit}.`,
          });
        }
      }
    }
  });

function readEye(
  values: FormValues,
  prefix: EyePrefix
): EyeValues | undefined {
  const eye: EyeValues = {};

  for (const measure of [
    "sph",
    "cyl",
    "axis",
  ] as const) {
    const text =
      values[`${prefix}_${measure}`].trim();

    if (text) {
      eye[measure] = Number(text);
    }
  }

  return Object.keys(eye).length ? eye : undefined;
}

// -----------------------------------------------------
// 1. LIGHTWEIGHT CUSTOMER CACHE
// -----------------------------------------------------

// nik - Keep recently viewed customer records in memory.
// This avoids showing a blank page every time staff go
// from Customer Details to Edit and back again.
//
// Data is always refreshed from the backend when
// the customer details page opens.
//
// Cache lives only in this browser tab's JavaScript memory.
const CACHE_TIME_MS = 30_000;
const MAX_CACHE_ITEMS = 20;

type CachedCustomer = {
  data: CustomerDetails;
  savedAt: number;
};

const customerCache = new Map<string, CachedCustomer>();

function readCachedCustomer(
  id: string
): CustomerDetails | null {
  const cached = customerCache.get(id);

  if (!cached) return null;

  if (Date.now() - cached.savedAt > CACHE_TIME_MS) {
    customerCache.delete(id);
    return null;
  }

  return cached.data;
}

function saveCachedCustomer(
  id: string,
  data: CustomerDetails
) {
  // Keep the cache small.
  if (customerCache.has(id)) {
    customerCache.delete(id);
  }

  customerCache.set(id, {
    data,
    savedAt: Date.now(),
  });

  if (customerCache.size > MAX_CACHE_ITEMS) {
    const oldestKey = customerCache.keys().next().value;

    if (oldestKey) {
      customerCache.delete(oldestKey);
    }
  }
}

function display(
  value: string | number | null | undefined
): string {
  if (
    value === null ||
    value === undefined ||
    value === ""
  ) {
    return "—";
  }

  return String(value);
}

// Format prices in Indian rupees.
function formatMoney(
  value: string | number | null | undefined
): string {
  if (
    value === null ||
    value === undefined ||
    value === ""
  ) {
    return "—";
  }

  const amount = Number(value);

  if (!Number.isFinite(amount)) {
    return "—";
  }

  return new Intl.NumberFormat("en-IN", {
    style: "currency",
    currency: "INR",
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  }).format(amount);
}

// Format a saved date without a timezone-related day shift.
function formatDate(
  value: string | null | undefined
): string {
  if (!value) return "Not recorded";

  const datePart = value.slice(0, 10);
  const match = /^(\d{4})-(\d{2})-(\d{2})$/.exec(datePart);

  if (!match) return "Unknown date";

  const year = Number(match[1]);
  const month = Number(match[2]);
  const day = Number(match[3]);

  const date = new Date(Date.UTC(year, month - 1, day));

  if (
    date.getUTCFullYear() !== year ||
    date.getUTCMonth() + 1 !== month ||
    date.getUTCDate() !== day
  ) {
    return "Unknown date";
  }

  return date.toLocaleDateString("en-IN", {
    day: "2-digit",
    month: "short",
    year: "numeric",
    timeZone: "UTC",
  });
}

// Format optical power with signs and two decimal places.
function formatPower(
  value: string | number | null | undefined
): string {
  if (
    value === null ||
    value === undefined ||
    value === ""
  ) {
    return "—";
  }

  const number = Number(value);

  if (!Number.isFinite(number)) {
    return "—";
  }

  const safeNumber = Object.is(number, -0) ? 0 : number;

  return `${safeNumber > 0 ? "+" : ""}${safeNumber.toFixed(2)}`;
}

// nik - Determine what was actually purchased from the
// saved product names. Pricing mode is NOT purchase type.
// Works for both catalog selections and custom typed names.
type PurchaseFlag =
  | "FRAME_ONLY"
  | "LENS_ONLY"
  | "FRAME_AND_LENS"
  | "NOT_SPECIFIED";

function getPurchaseFlag(record: CustomerRecord): PurchaseFlag {
  const hasFrame = Boolean(record.specsType?.trim());
  const hasLens = Boolean(record.lensType?.trim());

  if (hasFrame && hasLens) return "FRAME_AND_LENS";
  if (hasFrame) return "FRAME_ONLY";
  if (hasLens) return "LENS_ONLY";
  return "NOT_SPECIFIED";
}

// Use one consistent label and color across the page.
const purchaseFlagInfo: Record<
  PurchaseFlag,
  {
    label: string;
    color: "primary" | "success" | "info" | "default";
  }
> = {
  FRAME_ONLY: { label: "Frame Only", color: "info" },
  LENS_ONLY: { label: "Lens Only", color: "success" },
  FRAME_AND_LENS: { label: "Frame + Lens", color: "primary" },
  NOT_SPECIFIED: { label: "Not Specified", color: "default" },
};

// Display the product flag independently from the payment mode.
function PurchaseFlagChip({ record }: { record: CustomerRecord }) {
  const flag = purchaseFlagInfo[getPurchaseFlag(record)];
  return <Chip label={flag.label} color={flag.color} size="small" />;
}

  const [activeTab, setActiveTab] = useState(0);
  const [copyMessage, setCopyMessage] = useState("");

  // nik - Abort outdated customer requests when refreshing or switching customers.
  const requestRef = useRef<AbortController | null>(null);

  const previousSearch = params.get("search") ?? "";

  const searchQuery = previousSearch
    ? `?search=${encodeURIComponent(previousSearch)}`
    : "";

  const backUrl = `/customers${searchQuery}`;

  // Restore recent data immediately, then fetch fresh data.
  useEffect(() => {
    if (!id) {
      setCustomer(null);
      setLoading(false);
      setError("Customer ID is missing.");
      return;
    }

    requestRef.current?.abort();
    const controller = new AbortController();
    requestRef.current = controller;
    const cached = readCachedCustomer(id);

    setCustomer(cached);
    setLoading(!cached);
    setRefreshing(Boolean(cached));
    setError("");

    getCustomer(id, controller.signal)
      .then((result) => {
        if (controller.signal.aborted || requestRef.current !== controller || result.id !== id) return;

        saveCachedCustomer(id, result);
        setCustomer(result);
        setError("");
      })
      .catch((err) => {
        if (controller.signal.aborted || requestRef.current !== controller) return;

        setError(
          err instanceof Error
            ? err.message
            : "Unable to load customer details."
        );
      })
      .finally(() => {
        if (!controller.signal.aborted && requestRef.current === controller) {
          setLoading(false);

// Optical prescription measurements.
// Each value is optional because some prescriptions
// may not contain every measurement.
export type EyeValues = {
  sph?: number;
  cyl?: number;
  axis?: number;
};

// Reusable product information stored with a purchase.
// Catalog IDs are null when products are entered manually.
export type PurchaseProducts = {
  specsType: string | null;
  lensType: string | null;

  // nik - Link products to the Admin Catalog.
  // The backend validates these IDs before saving.
  frameCatalogId?: string | null;
  lensCatalogId?: string | null;
};

// Data sent when registering a new customer.
export type CustomerPayload = {
  name: string;
  mobNumber: string | null;
  gender: "Male" | "Female" | "Other" | null;
  reference: string | null;

  // Private information for shop staff only.
  notes: string | null;

  // Actual purchase date in YYYY-MM-DD format.
  // Null means the date has not been recorded.
  purchaseDate?: string | null;

  // Optical prescription is optional.
  prescription?: {
    re_dist?: EyeValues;
    re_near?: EyeValues;
    le_dist?: EyeValues;
    le_near?: EyeValues;
  };

  // nik - Support two pricing methods.
  // The backend calculates and validates final pricing.
  record:
    | (PurchaseProducts & {
        pricingMode: "SPLIT";
        frameRate: number;
        lensRate: number;
      })
    | (PurchaseProducts & {
        pricingMode: "MRP_ONLY";
        mrp: number;
      });
};

// Minimal response after successful customer creation.
export type CreatedCustomer = {
  id: string;
  name: string;
};

// Read JSON safely and display useful error messages.
async function readResponse<T>(
  response: Response
): Promise<T> {
  const result = await response.json().catch(() => null);

  if (!response.ok || !result?.success) {
    throw new Error(
      result?.message ||
        `Request failed (${response.status}). Please try again.`
    );
  }

  return result.data as T;
}

// nik - Send the Admin session cookie with every customer request.
// Keep this in one place instead of repeating it in each API function.
function apiFetch(url: string, options: RequestInit = {}) {
  return fetch(url, { ...options, credentials: "include" });
}

// Vite forwards /api requests to the Express server.
// No hardcoded backend address is required.
export async function createCustomer(
  payload: CustomerPayload
): Promise<CreatedCustomer> {
  const response = await apiFetch("/api/customers", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
    },
    body: JSON.stringify(payload),
  });

  const customer =
    await readResponse<CreatedCustomer>(response);

  // Confirm the backend returned a unique customer ID.
  if (!customer?.id) {
    throw new Error(
      "Customer was saved, but the server did not return a customer ID."
    );
  }

  return customer;
}

// 6. CUSTOMER SEARCH AND DETAILS

// Small customer summary shown in search results.
// Multiple customers may share the same mobile number.
export type CustomerSummary = {
  id: string;
  name: string;
  mobNumber: string | null;
  gender: string | null;
  createdAt: string;
};

// Complete customer information and history.
export type CustomerDetails = CustomerSummary & {
  reference: string | null;

  // Staff-only notes.
  // Never expose these in customer PDFs or WhatsApp.
  notes: string | null;

  updatedAt: string;

  prescriptions: Prescription[];
  records: CustomerRecord[];
};

// -----------------------------------------------------
// 7. SEARCH CUSTOMERS
// -----------------------------------------------------

// Search using a name or partial mobile number.
// Customer records are always identified by UUID.
export async function searchCustomers(
  search: string,
  signal?: AbortSignal
): Promise<CustomerSummary[]> {
  const value = search.trim();

  const params = new URLSearchParams();

  if (value) {
    if (/^\d+$/.test(value)) {
      params.set("phone", value);
    } else {
      params.set("name", value);
    }
  }

  const query = params.toString();

  const response = await apiFetch(
    `/api/customers${query ? `?${query}` : ""}`,
    {
      signal,
    }
  );

  return readResponse<CustomerSummary[]>(response);
}

// -----------------------------------------------------
// 8. GET CUSTOMER DETAILS
// -----------------------------------------------------

// Retrieve the selected customer by unique ID,
// never by their mobile number.
export async function getCustomer(
  id: string,
  signal?: AbortSignal
): Promise<CustomerDetails> {
  const response = await apiFetch(
    `/api/customers/${encodeURIComponent(id)}`,
    {
      signal,
    }
  );

  return readResponse<CustomerDetails>(response);
}

// 9. UPDATE CUSTOMER PERSONAL INFORMATION
// -----------------------------------------------------

export type CustomerPersonalUpdate = {
  name: string;
  mobNumber: string | null;
  gender: "Male" | "Female" | "Other" | null;
  reference: string | null;

  // Internal notes are editable by shop staff.
  notes: string | null;
};

// Update only the selected customer's personal details.
// Purchase history is edited through separate endpoints.
export async function updateCustomer(
  id: string,
  changes: CustomerPersonalUpdate
): Promise<CustomerDetails> {
  const response = await apiFetch(
    `/api/customers/${encodeURIComponent(id)}`,
    {
      method: "PATCH",
      headers: {
        "Content-Type": "application/json",
      },
      body: JSON.stringify(changes),
    }
  );

  return readResponse<CustomerDetails>(response);
}

// -----------------------------------------------------
// 12. UPDATE ONE HISTORICAL PURCHASE
// -----------------------------------------------------

// Update exactly one purchase using the customer ID
// and the selected purchase record ID.
// Other historical purchases remain unchanged.
export async function updatePurchase(
  customerId: string,
  recordId: string,
  changes: EditablePurchase
): Promise<CustomerRecord> {
  const response = await apiFetch(
    `/api/customers/${encodeURIComponent(customerId)}/records/${encodeURIComponent(recordId)}`,
    {
      method: "PATCH",
      headers: {
        "Content-Type": "application/json",
      },
      body: JSON.stringify(changes),
    }
  );

  return readResponse<CustomerRecord>(response);
}
